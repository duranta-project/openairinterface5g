# DSP side of the Hexagon LDPC offload: libldpc_hexagon_skel.so, loaded by
# FastRPC into the compute DSP.
#
# This is a separate CMake project, built by ../CMakeLists.txt as an
# ExternalProject, because it needs a different C compiler: CMake has one
# compiler per language per project, and this code is built with hexagon-clang
# (cmake_targets/cross-hexagon-cdsp.cmake) while the rest of OAI is built for
# the application processor.

cmake_minimum_required(VERSION 3.16)
project(ldpc_hexagon_skel C)

set(OAI_SOURCE_DIR "" CACHE PATH "OAI source tree")
set(HEXAGON_DSP_ARCH "v73" CACHE STRING "Hexagon architecture of the target CDSP (QCS9075: v73)")
option(LDPC_HEX_VTCM "Keep the decoder working set in VTCM (needs a signed protection domain)" ON)
option(LDPC_HEX_GATHER "Deinterleave with vgather from VTCM (falls back to scalar if VTCM is unavailable)" ON)
option(LDPC_HEX_SIMTEST "Build ldpc_hexagon_simtest, the decoder test run under hexagon-sim" OFF)

list(APPEND CMAKE_MODULE_PATH ${OAI_SOURCE_DIR}/cmake_targets/tools/MODULES)
find_package(HexagonSDK REQUIRED)

# shared libraries are what FastRPC loads, although the Generic platform does not declare them
set_property(GLOBAL PROPERTY TARGET_SUPPORTS_SHARED_LIBS TRUE)

set(_idl ${CMAKE_CURRENT_SOURCE_DIR}/../inc/ldpc_hexagon.idl)
add_custom_command(
  OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/ldpc_hexagon_skel.c ${CMAKE_CURRENT_BINARY_DIR}/ldpc_hexagon.h
  COMMAND ${HexagonSDK_QAIC} -mdll -o ${CMAKE_CURRENT_BINARY_DIR}
          -I${HexagonSDK_INCLUDE_DIR} -I${HexagonSDK_INCLUDE_DIR}/stddef ${_idl}
  DEPENDS ${_idl}
  VERBATIM
)

set(_hex_defs
  $<$<BOOL:${LDPC_HEX_VTCM}>:LDPC_HEX_VTCM>
  $<$<BOOL:${LDPC_HEX_GATHER}>:LDPC_HEX_GATHER>
)

add_library(ldpc_hexagon_skel SHARED
  ${CMAKE_CURRENT_BINARY_DIR}/ldpc_hexagon_skel.c
  ../src/ldpc_hexagon_imp.c
)
target_include_directories(ldpc_hexagon_skel PRIVATE
  ../inc
  ${CMAKE_CURRENT_BINARY_DIR} # ldpc_hexagon.h, generated by qaic
  ${OAI_SOURCE_DIR}/openair1/PHY/CODING/nrLDPC_decoder
  ${OAI_SOURCE_DIR}
)
target_compile_options(ldpc_hexagon_skel PRIVATE
  -m${HEXAGON_DSP_ARCH} -G0 -O2 -Wall -fpic -fno-zero-initialized-in-bss -fdata-sections
  -mhvx -mhvx-length=128B
)
target_compile_definitions(ldpc_hexagon_skel PRIVATE ${_hex_defs})

# FastRPC loads the skel into a user protection domain:
#  - G0/pic target libraries to match -G0 (mixing small-data models breaks
#    GP-relative addressing)
#  - -Bsymbolic, and the allocators wrapped onto the FastRPC-aware heap
#  - libc as a real DT_NEEDED entry, and a soname
set(_hexlib ${HexagonSDK_TOOLS_ROOT}/target/hexagon/lib)
target_link_options(ldpc_hexagon_skel PRIVATE
  -m${HEXAGON_DSP_ARCH} -G0 -fpic -shared
  -Wl,-Bsymbolic
  -Wl,-L${_hexlib}/${HEXAGON_DSP_ARCH}/G0/pic -Wl,-L${_hexlib}
  -Wl,--no-threads
  -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=free -Wl,--wrap=realloc -Wl,--wrap=memalign
  -Wl,-soname,libldpc_hexagon_skel.so
)
target_link_libraries(ldpc_hexagon_skel PRIVATE HexagonSDK::dsp c)
install(TARGETS ldpc_hexagon_skel LIBRARY DESTINATION .)

# The decoder alone, as a standalone program for hexagon-sim: needs no board.
# Registered as a test by ../CMakeLists.txt.
if(LDPC_HEX_SIMTEST)
  add_executable(ldpc_hexagon_simtest ../test/ldpc_hexagon_simtest.c)
  target_include_directories(ldpc_hexagon_simtest PRIVATE
    ../test/shim
    ../inc
    ${OAI_SOURCE_DIR}/openair1/PHY/CODING/nrLDPC_decoder
    ${OAI_SOURCE_DIR}/openair1/PHY/CODING/nrLDPC_encoder
    ${OAI_SOURCE_DIR}
  )
  target_compile_options(ldpc_hexagon_simtest PRIVATE -m${HEXAGON_DSP_ARCH} -O2 -Wall -mhvx -mhvx-length=128B)
  target_compile_definitions(ldpc_hexagon_simtest PRIVATE ${_hex_defs})
  target_link_options(ldpc_hexagon_simtest PRIVATE -m${HEXAGON_DSP_ARCH})
  target_link_libraries(ldpc_hexagon_simtest PRIVATE hexagon m)
  install(TARGETS ldpc_hexagon_simtest RUNTIME DESTINATION .)
endif()

##########################################################

# LDPC offload library - Qualcomm Hexagon compute DSP
##########################################################

find_package(HexagonSDK REQUIRED)

# FastRPC stub (ARM side) and the shared header, generated from the IDL
set(_idl ${CMAKE_CURRENT_SOURCE_DIR}/inc/ldpc_hexagon.idl)
add_custom_command(
  OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/ldpc_hexagon_stub.c ${CMAKE_CURRENT_BINARY_DIR}/ldpc_hexagon.h
  COMMAND ${HexagonSDK_QAIC} -mdll -o ${CMAKE_CURRENT_BINARY_DIR}
          -I${HexagonSDK_INCLUDE_DIR} -I${HexagonSDK_INCLUDE_DIR}/stddef ${_idl}
  DEPENDS ${_idl}
  VERBATIM
)

add_library(ldpc_hexagon MODULE
            src/nrLDPC_coding_hexagon.c
            src/ldpc_hexagon_arm.c
            ${CMAKE_CURRENT_BINARY_DIR}/ldpc_hexagon_stub.c
            ../../nrLDPC_encoder/ldpc_encoder_optim8segmulti.c
)
target_include_directories(ldpc_hexagon PRIVATE
                           inc
                           ${CMAKE_CURRENT_BINARY_DIR}) # ldpc_hexagon.h, generated by qaic
target_link_libraries(ldpc_hexagon PRIVATE ldpc_segment_common HexagonSDK::cdsprpc log_headers)
set_target_properties(ldpc_hexagon PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})

# DSP side, built with hexagon-clang. It lands next to libldpc_hexagon.so and
# has to be installed on the target where FastRPC looks for skels
# (e.g. /usr/lib/rfsa/adsp).
set(HEXAGON_DSP_ARCH "v73" CACHE STRING "Hexagon architecture of the target CDSP (QCS9075: v73)")
option(LDPC_HEX_VTCM "Keep the DSP decoder working set in VTCM (needs a signed protection domain)" ON)
option(LDPC_HEX_GATHER "Deinterleave with vgather from VTCM (falls back to scalar if VTCM is unavailable)" ON)
include(ExternalProject)
ExternalProject_Add(ldpc_hexagon_skel
  SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/dsp
  BINARY_DIR ${CMAKE_CURRENT_BINARY_DIR}/dsp
  CMAKE_ARGS
    -DCMAKE_TOOLCHAIN_FILE=${CMAKE_SOURCE_DIR}/cmake_targets/cross-hexagon-cdsp.cmake
    -DHEXAGON_TOOLS_ROOT=${HexagonSDK_TOOLS_ROOT}
    -DHEXAGON_SDK_ROOT=${HEXAGON_SDK_ROOT}
    -DOAI_SOURCE_DIR=${CMAKE_SOURCE_DIR}
    -DHEXAGON_DSP_ARCH=${HEXAGON_DSP_ARCH}
    -DLDPC_HEX_VTCM=${LDPC_HEX_VTCM}
    -DLDPC_HEX_GATHER=${LDPC_HEX_GATHER}
    -DLDPC_HEX_SIMTEST=${ENABLE_TESTS}
    -DCMAKE_INSTALL_PREFIX=${CMAKE_BINARY_DIR}
  BUILD_ALWAYS ON
)
add_dependencies(ldpc_hexagon ldpc_hexagon_skel)

add_dependencies(nr-softmodem ldpc_hexagon)
add_dependencies(nr_ulsim ldpc_hexagon)
add_dependencies(nr_ulschsim ldpc_hexagon)
add_dependencies(ldpctest ldpc_hexagon)

# The DSP decoder against OAI's encoder, run under hexagon-sim on the build
# host: BG1/BG2 at several lifting sizes and code rates, and HARQ combining.
# hexagon-sim exits 0 whatever the program returns, so judge by its summary.
if(ENABLE_TESTS)
  add_dependencies(tests ldpc_hexagon_skel)
  add_test(NAME ldpc_hexagon_sim
           COMMAND ${HexagonSDK_TOOLS_ROOT}/bin/hexagon-sim -m${HEXAGON_DSP_ARCH} ${CMAKE_BINARY_DIR}/ldpc_hexagon_simtest)
  set_tests_properties(ldpc_hexagon_sim PROPERTIES PASS_REGULAR_EXPRESSION "PASSED: 0 failure")
endif()

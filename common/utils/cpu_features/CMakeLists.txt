if(CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
  add_library(cpu_features OBJECT get_enabled_instructions.c cpu_check.c)
  set_source_files_properties(cpu_check.c PROPERTIES COMPILE_OPTIONS
    "-march=x86-64;-mno-avx;-mno-bmi;-mno-bmi2;-mno-lzcnt;-mno-movbe")
elseif(CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
  add_library(cpu_features OBJECT get_enabled_instructions.c cpu_check.c)
  set_source_files_properties(cpu_check.c PROPERTIES COMPILE_OPTIONS "-mgeneral-regs-only")
else()
  message("cpu_features: runtime CPU check only supported on `x86` and `arm`.")
endif()
